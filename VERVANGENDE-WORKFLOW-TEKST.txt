name: Update HandbalHub nieuws
on:
  workflow_dispatch:
  schedule:
    - cron: "17 */6 * * *"
  push:
    paths:
      - "update-news.mjs"
      - "sources.json"
      - "package.json"
      - ".github/workflows/main.yml"
permissions:
  contents: write
concurrency:
  group: handbalhub-news
  cancel-in-progress: true
jobs:
  update:
    runs-on: ubuntu-latest
    timeout-minutes: 4
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 24
      - run: npm run update:news
      - name: Geldige nieuwsupdate opslaan
        run: |
          git config user.name "HandbalHub Bot"
          git config user.email "actions@users.noreply.github.com"
          git add news.json
          git diff --cached --quiet && exit 0
          git commit -m "Automatische nieuwsupdate"
          git pull --rebase origin main
          git push origin HEAD:main
